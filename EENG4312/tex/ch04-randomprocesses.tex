\section{Chapter 4: Random Processes}

\subsection{Expected Value}
\label{subsec:expectedvalue}

\begin{idea}
	Expectation behaves like a ``center of mass’’ for a random variable.
	Each value is weighted by how plausible it is, and the result is the balance point.
\end{idea}

For a continuous random variable with pdf $f_X(x)$:
\begin{equation}\label{eq:expectedvalue}
	\mathbb{E}[X] = \int_{-\infty}^{\infty} x\, f_X(x)\, dx.
\end{equation}

For a discrete random variable, the integral becomes a sum:
\begin{equation}\label{eq:expectedvaluediscrete}
	\mathbb{E}[X] = \sum_x x\,P(X=x).
\end{equation}

One small notation nit-pick I have with the textbook. It often uses
\(\mathbb{E}[A+B]^2\) to mean \(\mathbb{E}[(A+B)^2]\). Where we are
getting the expectation of the square, not squaring the expectation.
Ideally we would write everything as \(\mathbb{E}[(A+B)^2]\), but if
the textbook is lazy with it, I will be lazy with it. Just keep in mind that
\begin{align*}
	\mathbb{E}[A+B]^2 & = \mathbb{E}[(A+B)^2]    \\
	\mathbb{E}[A+B]^2 & \neq (\mathbb{E}[A+B])^2
\end{align*}

\subsection{Uniform Distribution}

A uniform distribution over $[a,b]$ has constant density function:
\begin{equation}\label{eq:uniformpdf}
	f_X(x) =
	\begin{cases}
		\frac{1}{b-a} & a\leq x\leq b    \\
		0             & \text{otherwise}
	\end{cases}
\end{equation}

\begin{derivation}
	Compute the expected value:
	\[
		\mathbb{E}[X]
		= \int_a^b x \cdot \frac{1}{b-a}\, dx
		= \frac{1}{b-a}\int_a^b x\, dx
		= \frac{1}{b-a}\left[\frac{x^2}{2}\right]_{a}^{b}.
	\]
	Evaluating:
	\[
		\mathbb{E}[X] = \frac{b^2 - a^2}{2(b-a)} = \frac{a+b}{2}.
	\]
	Thus the mean of a uniform distribution is simply the midpoint.
\end{derivation}

\begin{idea}
	Symmetry around zero forces the mean to be zero without any calculation.
\end{idea}

\begin{examplebox}

	Using $a=-1$ and $b=1$:
	\[
		\mathbb{E}[X] = \frac{-1 + 1}{2} = 0.
	\]

\end{examplebox}

\subsection{Mean of a Random Process}


\begin{idea}
	The mean of a random process is the expected value applied
	pointwise in time. At each $t$, we average over all randomness
	that influences $X(t)$.
\end{idea}

For any process $X(t)$ constructed from random variables
and deterministic functions of $t$, linearity of expectation gives
\begin{equation}\label{eq:meanofprocess}
	m_X(t) = \mathbb{E}[X(t)].
\end{equation}

\begin{examplebox}[label=exa:uniform]
	Consider the process $X(t) = A + Bt$, where $A$ and $B$
	are independent random variables values uniform on \([-1,1]\).
	Using linearity:
	\[
		m_X(t) = \mathbb{E}[A] + t\,\mathbb{E}[B].
	\]
	For $A, B$ uniform on $[-1,1]$:
	\[
		\mathbb{E}[A] = \mathbb{E}[B] = 0,
	\]
	hence
	\[
		m_X(t) = 0.
	\]
\end{examplebox}

\subsection{Autocorrelation Function}
The autocorrelation function of the random process \(X(t)\) is denoted by
\(R_{XX}(t_1,t_2)\), which is often shortened to \(R_X(t_1,t_2)\) is defined as:
\begin{equation}\label{eq:autocorrelation}
	R_X(t_1,t_2) = \mathbb{E}[X(t_1)X(t_2)]
\end{equation}

From this definition we can see that the autocorrelation function can
be evaluated as:

\[
	R_X(t_1,t_2) = \int_{-\infty}^{\infty}
	\int_{-\infty}^{\infty}
	x_1 x_2 f_{X(t_1),X(t_2)}(x_1,x_2) dx_1 dx_2
\]

\begin{examplebox}
	Continuing on from \zcref{exa:uniform}. We can find the autocorrelation
	function for \(X(t) = A + Bt\) where \(A\) and \(B\) are independent random
	values uniform on \([-1,1]\).
	\begin{align*}
		R_X(t) & = \mathbb{E}[(A + Bt_1)(A + Bt_2)]                                               \\
		       & =\mathbb{E}[A^2] + \mathbb{E}[AB]t_2 + \mathbb{E}[AB]t_1 + \mathbb{E}[B^2]t_1t_2
	\end{align*}

	Because \(A\) and \(B\) are independent random variables we know that

	\[
		\mathbb{E}[AB] = \mathbb{E}[A]\mathbb{E}[B] = 0
	\]

	We can also compute \(\mathbb{E}[A^2]\) and \(\mathbb{E}[B^2]\) as follows:

	\[
		\mathbb{E}[A^2] = \int_{-1}^{1} x^2 \frac{1}{2} dx
		= \frac{1}{6} x^3 \Big|_{-1}^{1} = \frac{1}{3}
	\]

	Where we use the pdf of a uniform random variable over \([-1,1]\) which is \(\frac{1}{2}\)
	(See~\zcref{eq:uniformpdf}). Tying this all together we have:

	\[
		R_X(t_1,t_2) = \frac{1}{3} + \frac{1}{3} t_1 t_2
	\]

\end{examplebox}

\subsection{Wide-Sense Stationary (WSS) Random Process}
A process \(X(t)\) is said to be wide-sense stationary (WSS) if the following
conditions are satisfied:
\begin{enumerate}
	\item The mean \(m_X(t) = \mathbb{E}[X(t)]\) is independent of \(t\).
	\item The autocorrelation function \(R_X(t_1,t_2)\) depends only on the time
	      difference \(\tau = t_1 - t_2\), and not on \(t_1\) and \(t_2\) individually.
	      \[
		      R_X(t_1,t_2) = R_X(t_1 - t_2) = R_X(\tau)
	      \]
\end{enumerate}

This means for WSS processes, the definition of the autocorrelation function
can be simplified to:
\[
	R_X(t+\tau,t) = R_X(\tau) = \mathbb{E}[X(t)X(t-\tau)]
\]

In general, the term \textit{stationary} is used to refer to WSS processes.
For these processes, the mean is often denoted simply as \(m_X\) and the
autocorrelation function as \(R_X(\tau)\).

The autocorrelation function of the stationary process has the following properties:
\begin{enumerate}
	\item \(R_X(\tau)\) is an even function; i.e., \(R_X(-\tau)=R_X(\tau)\)
	      \begin{derivation}
		      By the definition of the autocorrelation function:
		      \[
			      R_X(\tau)=\mathbb{E}[X(t)X(t-\tau)]=\mathbb{E}[X(t-\tau)X(t)]=R_X(-\tau)
		      \]
	      \end{derivation}
	\item The maximum absolute value of \(R_X(\tau)\) is achieved at \(\tau=0\);\\
	      i.e., \(\abs{R_X(\tau)} \leq R_X(0)\).
	      \begin{derivation}
		      I'm going to first give you the textbook's proof. Then give you another
		      proof with the Cauchy-Schwarz inequality.
		      It is important to note that for WSS processes
		      \[
			      R_X(0)=\mathbb{E}[X(t)X(t)]=\mathbb{E}[X^2(t)]
		      \]
		      Since squares of anything are always positive, we can note that
		      \[
			      \mathbb{E}[X(t) \pm X(t-\tau)]^2 \geq 0
		      \]
		      We can then expand this to get
		      \[
			      \mathbb{E}[X^2(t)]+\mathbb{E}[X^2(t-\tau)]
			      \pm 2\mathbb{E}[X(t)X(t-\tau)] \geq 0
		      \]
		      and with some algebraic manipulation we can get to:
		      \begin{align*}
			      \mathbb{E}[X^2(t)]+\mathbb{E}[X^2(t-\tau)]
			                      & \pm 2\mathbb{E}[X(t)X(t-\tau)] \geq 0      \\
			      R_X(0) + R_X(0) & \pm 2 R_X(\tau) \geq 0                     \\
			                      & \pm 2R_X(\tau) \geq -2R_X(0)               \\
			                      & \pm R_X(\tau)                  \leq R_X(0)
		      \end{align*}
		      which means \(\abs{R_X(\tau)} \leq R_X(0)\).

		      Now onto the Cauchy-Scharz Version: \\
		      Keep in mind we are trying to show that \(\abs{R_X(\tau)} \leq R_X(0)\).
		      We should briefly introduce the Cauchy–Schwarz inequality:
		      \[
			      \abs{\mathbb{E}[UV]} \leq \sqrt{\mathbb{E}[U^2]\mathbb{E}[V^2]}
		      \]
		      This little inequality actually makes the proof shorter.
		      By setting \(U=X(t)\) and \(V=X(t-\tau)\):
		      \[
			      \abs{R_X(\tau)} = \abs{\mathbb{E}[X(t)X(t-\tau)]}
			      \leq \sqrt{\mathbb{E}[X^2(t)]\mathbb{E}[X^2(t-\tau)]}
			      = R_X(0)
		      \]
	      \end{derivation}
	\item If for some \(T_0\) we have \(R_X(T_0)= R_X(0)\), then for all integers \(k\),\\
	      \(R_X(kT_0)=R_X(0)\)
	      \begin{idea}
		      If an autocorrelation function reaches its maximum
		      \(R_X(0)\) at some non-zero lag \(T_0\), then it must equal \(R_X(0)\)
		      again at every integer multiple of that lag.
	      \end{idea}

	      \begin{derivation}
		      I'm gonna be honest, I don't feel like translating their proof into
		      something I understand right now. It's on page 169 of the textbook
		      though.

		      Here are the first steps for it though (I got scared of the verbosity
		      approximately halfway down):

		      We want to show:
		      \[
			      R_X(kT_0)=R_X(0)\ \text{for all integers}\ k
		      \]
		      The inductive hypothesis is \(R_X(iT_0) = R_X(0)\), for \(i = 1,2,...,k-1\).

		      Starting with a non-negative square:
		      \[
			      \mathbb{E}[X(t)-X(t-kT_0)]^2 \geq 0
		      \]
		      We can then split the shift \(kT_0\) into two smaller shifts:
		      \[
			      kT_0 = kT_0 - T_0 + T_0 = (k-1)T_0 + T_0
		      \]

		      So we can then expand our non-negative square like so:
		      \begin{align}
			       & \mathbb{E}[X(t)-X(t-kT_0)]^2 =  \notag \\
			       & \mathbb{E}[(X(t)-X(t-(k-1)T_0)) \notag \\
			       & +(X(t-(k-1)T_0)-X(t-kT_0))]^2
			      \label{eq:nonnegative-square}
		      \end{align}

		      We can also expand things in a different way. When we expand:
		      \[
			      \mathbb{E}[X(t)-X(t-kT_0)]^2
		      \]
		      We end up with the following
		      (\textit{Remember my notation nit-pick at the end of \zcref{subsec:expectedvalue}}):
		      \[
			      \mathbb{E}[X^2(t)] + \mathbb{E}[X^2(t-kT_0)]
			      - 2\mathbb{E}[X(t)X(t-kT_0)]
		      \]
		      Which can be simplified to:
		      \[
			      R_X(0) + R_X(0) - 2R_X(kT_0) = 2R_X(0) - 2R_X(kT_0)
		      \]
	      \end{derivation}
\end{enumerate}

\subsection{Cyclostationary Random Processes}
New class of processes just dropped: \textit{cyclostationary processes}.

\begin{idea}
	The statistical properties of cyclostationary processes are periodic with time.
\end{idea}

A process \(X(t)\) is called cyclostationary if its statistics repeat periodically in time.
That means the mean satisfies
\[
	m_X(t+T_0)=m_X(t)
\]
and the autocorrelation also repeats every period:
\[
	R_X(t+\tau+T_0,t+T_0)=R_X(t+\tau,t)
\]

In other words, the process behaves like a stationary one, but only once per cycle.

For a cyclostationary process, we define the average
autocorrelation function \(\bar{R}_X(\tau)\), as the average of \(R_X(t+\tau,t)\).
\[
	\bar{R}_X(\tau)=\frac{1}{T_0}\int_0^{T_0}R_X(t+\tau,t)dt
\]
